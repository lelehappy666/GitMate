# GitMate 本地优先仓库与封面懒加载实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将页面 10–11 改为本地仓库优先，增加按需云端分页，并实现视口封面懒加载、持久缓存、限流停止和布局修复。

**Architecture:** 同步偏好、仓库分类、云端分页和封面调度分别由独立核心组件负责。工作台只接收本地管理仓库；云端分页只在用户打开时加载；封面解析与下载由视口调度器按可见集合和下一行驱动，并通过磁盘缓存复用。

**Tech Stack:** Swift 6、SwiftUI、Observation、Foundation、URLSession、ImageIO、CryptoKit、UserDefaults、自定义异步测试运行器、macOS 14

## Global Constraints

- 所有文档和 Git 提交使用中文。
- 工作台和默认仓库页只管理本地仓库。
- 云端仓库不计入本地健康度、异常数量和今日焦点。
- 云端列表每页 30 个仓库。
- 封面只加载当前可见卡片和下一行，最大并发数为 3。
- 正常浏览优先读取磁盘封面，不重复请求 README 或图片。
- 手动刷新只标记封面需要检查，不一次刷新全部封面。
- 封面固定为 `2:3`，不得参与网格列宽计算。
- 收到 GitHub 限流后停止启动新的在线请求。
- 修改全部完成后运行完整测试、Debug/Release 构建、原生界面回归和应用打包验证。

在新文件定义：

```swift
public protocol RepositorySyncPreferenceStoring: Sendable {
    func load(accountID: String) throws -> [RepositorySyncPreference]
    func save(
        _ preferences: [RepositorySyncPreference],
        accountID: String
    ) throws
    func clear(accountID: String) throws
}

public final class UserDefaultsRepositorySyncPreferenceStore:
    RepositorySyncPreferenceStoring,
    @unchecked Sendable
{
    private let defaults: UserDefaults
    private let lock = NSLock()

    public init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    public func load(accountID: String) throws -> [RepositorySyncPreference] {
        lock.lock()
        defer { lock.unlock() }
        guard let data = defaults.data(forKey: key(accountID)) else { return [] }
        return try JSONDecoder().decode(
            [RepositorySyncPreference].self,
            from: data
        )
    }

    public func save(
        _ preferences: [RepositorySyncPreference],
        accountID: String
    ) throws {
        lock.lock()
        defer { lock.unlock() }
        let normalized = Dictionary(
            preferences.map { ($0.repositoryID, $0) },
            uniquingKeysWith: { _, newest in newest }
        ).values.sorted { $0.repositoryID < $1.repositoryID }
        defaults.set(
            try JSONEncoder().encode(normalized),
            forKey: key(accountID)
        )
    }

    public func clear(accountID: String) throws {
        lock.lock()
        defer { lock.unlock() }
        defaults.removeObject(forKey: key(accountID))
    }

    private func key(_ accountID: String) -> String {
        "GitMate.RepositorySyncPreferences.\(Data(accountID.utf8).base64EncodedString())"
    }
}
```

向 `OnboardingDependencies` 注入 `repositorySyncPreferenceStore`。生产环境使用 `UserDefaultsRepositorySyncPreferenceStore()`，测试使用内存实现。

private final class InMemoryRepositorySyncPreferenceStore:
    RepositorySyncPreferenceStoring,
    @unchecked Sendable
{
    private var values: [String: [RepositorySyncPreference]]

    init(values: [String: [RepositorySyncPreference]] = [:]) {
        self.values = values
    }

    func load(accountID: String) throws -> [RepositorySyncPreference] {
        values[accountID] ?? []
    }

    func save(
        _ preferences: [RepositorySyncPreference],
        accountID: String
    ) throws {
        values[accountID] = preferences
    }

    func clear(accountID: String) throws {
        values[accountID] = nil
    }

```swift
public struct WorkspaceRepositoryGroups: Equatable, Sendable {
    public let local: [Repository]
    public let cloud: [Repository]
}

public enum WorkspaceRepositoryClassifier {
    public static func classify(
        repositories: [Repository],
        records: [LocalRepositoryRecord],
        preferences: [RepositorySyncPreference]
    ) -> WorkspaceRepositoryGroups {
        let recordsByID = Dictionary(
            uniqueKeysWithValues: records.map { ($0.repository.id, $0) }
        )
        let modesByID = Dictionary(
            uniqueKeysWithValues: preferences.map { ($0.repositoryID, $0.mode) }
        )
        let local = repositories.filter { repository in
            let availability = recordsByID[repository.id]?.availability ?? .missing
            return availability != .missing
                || (modesByID[repository.id] ?? .never) != .never
        }
        let localIDs = Set(local.map(\.id))
        return WorkspaceRepositoryGroups(
            local: local,
            cloud: repositories.filter { !localIDs.contains($0.id) }
        )
    }
}
```

- [ ] **Step 4: 实现限流门**

```swift
public actor WorkspaceRateLimitGate {
    private var resetAt: Date?

    public func check(now: Date) throws {
        guard let resetAt else { return }
        if now >= resetAt {
            self.resetAt = nil
            return
        }
        throw WorkspaceAPIError.rateLimited(resetAt: resetAt)
    }

    public func pause(until date: Date) {
        resetAt = max(resetAt ?? date, date)
    }
}
```

每次启动在线摘要、README 或封面解析前检查门。收到 `.rateLimited(resetAt:)` 后暂停门，取消任务组尚未开始的任务，并停止添加新任务。

- [ ] **Step 9: 实现工作区单仓库下载**

`syncRepositoryFromWorkspace` 验证账户、仓库和模式，拒绝 `.never`，保存目标偏好后复用 `runSync`：

```swift
public func syncRepositoryFromWorkspace(
    repository: Repository,
    mode: RepositorySyncMode
) async {
    guard mode != .never else {
        state.errorMessage = "请选择需要下载的云端仓库。"
        return
    }
    if !state.repositories.contains(where: { $0.id == repository.id }) {
        state.repositories.append(repository)
    }
    let preference = RepositorySyncPreference(
        repositoryID: repository.id,
        mode: mode
    )
    updateSyncMode(repositoryID: repository.id, mode: mode)
    state.failedRepositoryIDs.removeAll { $0 == repository.id }
    state.route = .syncProgress
    await runSync(repositories: [repository], preferences: [preference])
}
```

云端分页得到的仓库可能尚未出现在会话缓存中，因此入口必须接收完整
`Repository`，不能只按 ID 在旧列表中查找。`runSync` 成功时现有
`.finished` 事件会回到 `.complete`；失败、断网或授权失效时保留对应异常页面，
不得强制覆盖路由。
